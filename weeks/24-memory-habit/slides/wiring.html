[regions."hippocampus.episodic_control"] d = 256 gamma = 0.995 # fade per outcome kappa = 1.0 # act above one sigma [regions."basal_ganglia.striatum"] eta = 0.1 [inputs] "basal_ganglia.striatum.inhibit" = "hippocampus.episodic_control.confident" "control.handover.memory" = "...episodic_control.choice" "control.handover.habit" = "basal_ganglia.striatum.choice"

confident: memory's margin clears its noise; it holds the striatum's race.

handover: memory's move if confident, the striatum's otherwise.

Both learn from every outcome, on the move actually taken.
