from ubrain import Brain, GaussianCodes camera = GaussianCodes(dim=64, seed=1) # the frozen model itself: any codec brain = Brain(variant="small") brain.attach("sense", "vision", codec=camera) # -> key "sense.vision" brain.attach("reward") # -> key "reward" brain.attach("action", n=4) # -> "action" in the output r = None for t in range(3000): p = world.picture() a = brain.tick({"sense.vision": p, "reward": r})["action"] r = world.pay(p, a) # 1 if a is right for p
